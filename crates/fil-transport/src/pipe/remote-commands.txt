# The remote commands fil runs (DESIGN.md section 8.2, docs/m4/plan.md section 2.5).
# Pinned by fil-transport's pipe::command tests; FIL_UPDATE_GOLDEN=1 rewrites it after a
# deliberate change, which needs the owner's sign-off (CLAUDE.md, the section 11 invariants).
# The bootstrap is shown for version 0.1.0 and wire 1.12; each build holds its own.
plain = fil server --stdio
bootstrap = sh -c 'v=0.1.0-w1.12; w="(wire 1.12)"; u=$(id -u); b="$HOME/.local/share/fil"; f="$HOME/.local/bin/fil"; n=""; k1=""; k2=""; k3=""; say() { printf "%s" "$*"; echo; }; mac() { [ "$(uname -s)" = Darwin ] || return 1; ls -lde "$1" | grep -q " allow "; }; ok() { [ -d "$1" ] || return 1; [ -h "$1" ] && return 1; set -- "$1" $(ls -ldn "$1"); case $2 in d????-??-?*) ;; *) return 1;; esac; [ "$4" = "$u" ] || [ "$4" = 0 ] || return 1; mac "$1" && return 1; return 0; }; own() { [ -d "$1" ] || return 1; [ -h "$1" ] && return 1; set -- "$1" $(ls -ldn "$1"); case $2 in drwx------|drwx------[.@]) ;; *) return 1;; esac; [ "$4" = "$u" ] || return 1; mac "$1" && return 1; return 0; }; exe() { [ -f "$1" ] || return 1; [ -h "$1" ] && return 1; set -- "$1" $(ls -ln "$1"); case $2 in -????-??-?|-????-??-?[.@]) ;; *) return 1;; esac; [ "$4" = "$u" ] && [ -x "$1" ] || return 1; mac "$1" && return 1; return 0; }; chain() { ok "$HOME" && ok "$HOME/.local" && ok "$HOME/.local/share" && own "$b"; }; lb() { ok "$HOME" && ok "$HOME/.local" && ok "$HOME/.local/bin" && exe "$f"; }; pre() { [ -e "$1" ] || [ -h "$1" ] || return 0; "$2" "$1"; }; undo() { [ -n "$n" ] && rm -rf "$n"; [ -n "$k3" ] && rmdir "$b"; [ -n "$k2" ] && rmdir "$HOME/.local/share"; [ -n "$k1" ] && rmdir "$HOME/.local"; exit 1; }; IFS= read -r m || exit 1; if [ "$m" = check ]; then if chain; then for d in "$b"/*; do own "$d" && exe "$d/fil" && say FIL-HAVE "${d##*/}"; done; fi; command -v fil >/dev/null && say FIL-PATH "$(fil --version </dev/null | head -n 1)"; lb && say FIL-LOCAL "$("$f" --version </dev/null | head -n 1)"; exit 0; fi; if [ "$m" = remove ]; then if chain && own "$b/$v" && exe "$b/$v/fil"; then say FIL-BOOT1 gc; exec "$b/$v/fil" server --install-gc; fi; say FIL-BOOT1 none; exit 1; fi; [ "$m" = attach ] || exit 1; if chain && own "$b/$v" && exe "$b/$v/fil"; then say FIL-BOOT1 exec; exec "$b/$v/fil" server --stdio; fi; if lb && "$f" --version </dev/null | grep -qF "$w"; then say FIL-BOOT1 exec; exec "$f" server --stdio; fi; say FIL-BOOT1 attach "$(uname -s)" "$(uname -m)"; IFS=" " read -r c s h || exit 1; [ "$c" = put ] || exit 0; [ -n "$s" ] && [ "${#s}" -le 9 ] && [ "$(LC_ALL=C expr "x$s" : "x[0-9]*")" = "$((${#s}+1))" ] && [ "$s" -le 134217728 ] && [ "${#h}" -eq 64 ] && [ "$(LC_ALL=C expr "x$h" : "x[0-9a-f]*")" = 65 ] || exit 1; if command -v sha256sum >/dev/null; then t=sha256sum; elif command -v shasum >/dev/null; then t="shasum -a 256"; else say FIL-BOOT1 nohash; exit 1; fi; ok "$HOME" && pre "$HOME/.local" ok && pre "$HOME/.local/share" ok && pre "$b" own || { say FIL-BOOT1 baddir; exit 1; }; [ -d "$HOME/.local" ] || { mkdir -m 700 "$HOME/.local" || undo; k1=1; }; [ -d "$HOME/.local/share" ] || { mkdir -m 700 "$HOME/.local/share" || undo; k2=1; }; [ -d "$b" ] || { mkdir -m 700 "$b" || undo; k3=1; }; chain || { say FIL-BOOT1 baddir; undo; }; mkdir -m 700 "$b/.new-$$" || undo; n="$b/.new-$$"; ( umask 077; head -c "$s" > "$n/fil" ) || undo; z=$(wc -c < "$n/fil") || undo; [ "$(($z))" -eq "$s" ] || undo; x=$($t < "$n/fil"); [ "${x%% *}" = "$h" ] || { say FIL-BOOT1 badhash; undo; }; chmod 700 "$n/fil" || undo; exec "$n/fil" server --install-finish "$v" "$h" "$k1$k2$k3"'
